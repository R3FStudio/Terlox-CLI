add_library(terlox_installation STATIC
  model/InstallPlan.cpp model/InstallPlan.h
  backend/InstallEngine.cpp backend/InstallEngine.h
  uninstall/UninstallPlan.cpp uninstall/UninstallPlan.h uninstall/UninstallEngine.cpp
  integration/DesktopIntegration.cpp integration/DesktopIntegration.h)
target_link_libraries(terlox_installation PUBLIC Qt6::Core terlox_defaults)
target_include_directories(terlox_installation PUBLIC ${PROJECT_SOURCE_DIR})
set(setup_ui_target terlox_setup)
if(WIN32)
  set_source_files_properties(../branding/Terlox.rc PROPERTIES OBJECT_DEPENDS "${PROJECT_SOURCE_DIR}/branding/icons/terlox.ico")
  target_sources(terlox_installation PRIVATE windows/InstalledApplication.cpp windows/RunningFiles.cpp windows/RemoveIntegration.cpp windows/RepairUninstaller.cpp)
  target_link_libraries(terlox_installation PRIVATE rstrtmgr shell32 ole32)
  add_executable(terlox_maintenance maintenance/main.cpp)
  set_target_properties(terlox_maintenance PROPERTIES OUTPUT_NAME terlox-maintenance)
  target_link_libraries(terlox_maintenance PRIVATE terlox_installation)
  target_sources(terlox_maintenance PRIVATE ../branding/Terlox.rc ${CMAKE_BINARY_DIR}/branding/Version.rc)
  target_include_directories(terlox_maintenance PRIVATE ${PROJECT_SOURCE_DIR}/branding)
  set(setup_ui_target terlox_setup_wizard)
  add_executable(terlox_setup WIN32 windows/Bootstrap.cpp ../branding/Terlox.rc ${CMAKE_BINARY_DIR}/branding/Version.rc)
  set_target_properties(terlox_setup PROPERTIES OUTPUT_NAME terlox-setup AUTOMOC OFF MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
  target_include_directories(terlox_setup PRIVATE ${PROJECT_SOURCE_DIR}/branding)
  target_link_libraries(terlox_setup PRIVATE shell32 user32)
  target_compile_definitions(terlox_setup PRIVATE TERLOX_SETUP_VERSION="${PROJECT_VERSION}")
endif()
add_executable(${setup_ui_target} WIN32 main.cpp wizard/SetupWizard.cpp wizard/SetupWizard.h wizard/SetupPages.cpp wizard/SetupPages.h)
if(WIN32)
  set_target_properties(${setup_ui_target} PROPERTIES OUTPUT_NAME terlox-setup-wizard)
  add_dependencies(terlox_setup ${setup_ui_target})
else()
  set_target_properties(${setup_ui_target} PROPERTIES OUTPUT_NAME terlox-setup)
endif()
target_link_libraries(${setup_ui_target} PRIVATE terlox_installation terlox_branding Qt6::Widgets)
if(WIN32)
  target_include_directories(${setup_ui_target} PRIVATE ${PROJECT_SOURCE_DIR}/branding)
  target_sources(${setup_ui_target} PRIVATE ../branding/Terlox.rc ${CMAKE_BINARY_DIR}/branding/Version.rc)
  install(TARGETS ${setup_ui_target} RUNTIME DESTINATION bin)
elseif(APPLE)
  set_target_properties(terlox_setup PROPERTIES MACOSX_BUNDLE TRUE MACOSX_BUNDLE_GUI_IDENTIFIER "app.terlox.setup" MACOSX_BUNDLE_BUNDLE_NAME "Terlox Setup" MACOSX_BUNDLE_ICON_FILE "terlox.icns")
  target_sources(terlox_setup PRIVATE ../branding/icons/terlox.icns)
  set_source_files_properties(../branding/icons/terlox.icns PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
endif()
if(WIN32)
  install(TARGETS terlox_setup RUNTIME DESTINATION bin)
endif()
