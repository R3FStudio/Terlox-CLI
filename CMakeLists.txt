cmake_minimum_required(VERSION 3.24)
project(Terlox VERSION 1.0.0 LANGUAGES C CXX)
if(WIN32)
  enable_language(RC)
  configure_file(branding/Version.rc.in ${CMAKE_BINARY_DIR}/branding/Version.rc @ONLY)
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_C_STANDARD 11)
set(CMAKE_AUTOMOC ON)
set(CMAKE_RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
find_package(Qt6 6.8 REQUIRED COMPONENTS Core Gui Widgets Quick QuickWidgets Test Network)
include(CTest)
if(MSVC)
  add_compile_options(/utf-8 /W3)
  add_compile_definitions(_CRT_SECURE_NO_WARNINGS NOMINMAX WIN32_LEAN_AND_MEAN _WIN32_WINNT=0x0A00)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
include(cmake/Dependencies.cmake)
add_subdirectory(branding)
add_subdirectory(information)
add_subdirectory(updates)
add_subdirectory(integration)
add_subdirectory(shell)
add_subdirectory(terminal)
add_subdirectory(platform)
add_subdirectory(settings)
add_subdirectory(sessions)
add_subdirectory(renderer)
add_subdirectory(window)
add_subdirectory(app)
add_subdirectory(installer)
option(TERLOX_BUILD_PRESENTATION "Build the developer screenshot tool" OFF)
if(TERLOX_BUILD_PRESENTATION)
  add_subdirectory(branding/presentation)
endif()
if(BUILD_TESTING)
  add_subdirectory(tests)
endif()
install(TARGETS terlox terloxsh RUNTIME DESTINATION bin BUNDLE DESTINATION .)
install(FILES LICENSE NOTICE DESTINATION share/terlox/licenses)
install(FILES dependencies/libvterm/LICENSE DESTINATION share/terlox/licenses RENAME libvterm-MIT.txt)
install(FILES dependencies/fonts/cascadia/LICENSE DESTINATION share/terlox/licenses RENAME Cascadia-OFL-1.1.txt)
