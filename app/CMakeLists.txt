add_executable(terlox WIN32 main.cpp Application.cpp Application.h LaunchArguments.cpp LaunchArguments.h)
target_link_libraries(terlox PRIVATE terlox_window terlox_branding terlox_defaults)
if(WIN32)
  target_link_libraries(terlox PRIVATE terlox_handoff)
  target_include_directories(terlox PRIVATE ${PROJECT_SOURCE_DIR}/branding)
  target_sources(terlox PRIVATE ../branding/Terlox.rc ${CMAKE_BINARY_DIR}/branding/Version.rc)
  set_source_files_properties(../branding/Terlox.rc PROPERTIES INCLUDE_DIRECTORIES "${PROJECT_SOURCE_DIR}/branding" OBJECT_DEPENDS "${PROJECT_SOURCE_DIR}/branding/icons/terlox.ico")
endif()

if(APPLE)
  target_sources(terlox PRIVATE FileOpenHandler.cpp FileOpenHandler.h)
  target_sources(terlox PRIVATE ../branding/icons/terlox.icns)
  set_source_files_properties(../branding/icons/terlox.icns PROPERTIES MACOSX_PACKAGE_LOCATION Resources)
  set_target_properties(terlox PROPERTIES MACOSX_BUNDLE_INFO_PLIST "${PROJECT_SOURCE_DIR}/integration/macos/Info.plist.in")
  set_target_properties(terlox PROPERTIES MACOSX_BUNDLE TRUE MACOSX_BUNDLE_GUI_IDENTIFIER "app.terlox.terminal" MACOSX_BUNDLE_BUNDLE_NAME "Terlox" MACOSX_BUNDLE_SHORT_VERSION_STRING "${PROJECT_VERSION}")
  add_dependencies(terlox terloxsh)
  add_custom_command(TARGET terlox POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy $<TARGET_FILE:terloxsh> $<TARGET_BUNDLE_DIR:terlox>/Contents/MacOS/terloxsh)
  add_dependencies(terlox terloxctl)
  add_custom_command(TARGET terlox POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy $<TARGET_FILE:terloxctl> $<TARGET_BUNDLE_DIR:terlox>/Contents/MacOS/terloxctl)
