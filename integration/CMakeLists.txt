if(UNIX AND NOT APPLE)
  install(PROGRAMS linux/set-default.py DESTINATION share/terlox/integration)
  install(FILES linux/app.terlox.terminal.desktop DESTINATION share/applications)
  install(FILES assets/app.terlox.terminal.svg DESTINATION share/icons/hicolor/scalable/apps)
  foreach(size 16 24 32 48 64 128 256 512 1024)
    install(FILES "${PROJECT_SOURCE_DIR}/branding/icons/terlox-${size}.png" DESTINATION "share/icons/hicolor/${size}x${size}/apps" RENAME app.terlox.terminal.png)
  endforeach()
endif()
add_library(terlox_defaults STATIC defaults/DefaultTerminal.h defaults/Command.h defaults/Command.cpp)
target_include_directories(terlox_defaults PUBLIC ${PROJECT_SOURCE_DIR})
target_link_libraries(terlox_defaults PUBLIC Qt6::Core)
if(WIN32)
  target_sources(terlox_defaults PRIVATE windows/DefaultTerminal.cpp)
  target_link_libraries(terlox_defaults PRIVATE advapi32 ole32)
  add_subdirectory(windows/handoff)
elseif(APPLE)
  enable_language(OBJCXX)
  set_property(TARGET terlox_defaults PROPERTY OBJCXX_STANDARD 20)
  target_sources(terlox_defaults PRIVATE macos/DefaultTerminal.mm)
  target_link_libraries(terlox_defaults PRIVATE "-framework CoreServices")
else()
  target_sources(terlox_defaults PRIVATE linux/DefaultTerminal.cpp)
endif()
add_executable(terloxctl cli/main.cpp)
target_link_libraries(terloxctl PRIVATE terlox_defaults terlox_updates)
install(TARGETS terloxctl RUNTIME DESTINATION bin)
